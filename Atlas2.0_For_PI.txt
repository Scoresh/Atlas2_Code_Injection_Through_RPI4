void loop() {
  if (Serial.available() > 0) {
    String data = Serial.readStringUntil('\n');
    parseCommand(data);
  }
}

void parseCommand(String cmd) {
  if (cmd.startsWith("FORM")) {
//    int targetMode = cmd.substring(5).toInt();
//    changeForm(targetMode);
  }
  else if (cmd.startsWith("M")) {
   // if (currentMode == 1) accelerate(cmd.substring(2).toFloat());
    Serial.println("Moving Straight");
    straight();
    accelerate(cmd.substring(2).toFloat());
   
  }
  else if (cmd.startsWith("T")) {
    //if (currentMode == 2) accelerate(cmd.substring(2).toFloat());
    Serial.println("Turning");
    //angleWheels();
    executePointTurn(cmd.substring(2).toFloat());
  }
  else if (cmd.startsWith("L")) {
    Serial.println("Move drum 1");
    stopActuators();
    move_drum(1);
  }
  else if (cmd.startsWith("R")) {
    Serial.println("Move drum 2");
    stopActuators();
    move_drum(2);
  }
  else if (cmd.startsWith("A")) {
    Serial.println("Move drum 0");
    move_drum(0);
  }
  else if (cmd.startsWith("D")) {
    Serial.println("Mining");
    mining = true;
    depositing = false;
    stopMotors();
    run_drum(1, cmd.substring(2).toFloat());
  }
  else if (cmd.startsWith("P")) {
    Serial.println("Depositing");
    mining = false;
    depositing = true;
    stopMotors();
    run_drum(2, 1.0);
  }
  /*else if (cmd.startsWith("READ")) {
    if (checkCapacity()) {
      Serial.println("Full");

    }
    else {
      Serial.println("Continue");
    }
  }*/
  else {
    Serial.println("stopping");
    stopMotors();
    run_drum(3, 0);
    stopActuators();
  }
}

void angleWheels(){
  if(limit1==HIGH || limit2 == HIGH){
    return;
  }
  stopMotors();
  while(limit1==LOW && limit2 == LOW){
    digitalWrite(D_ACT_POS,LOW);
    digitalWrite(D_ACT_NEG,HIGH);
  }
  digitalWrite(D_ACT_NEG,LOW);
  currentMode = 2;
}

void straight(){
  if(limit1 ==LOW && limit2 == LOW){
    return;
  }
  stopMotors();
  digitalWrite(D_ACT_NEG,LOW);
  digitalWrite(D_ACT_POS,HIGH);
  delay(10000);
  digitalWrite(D_ACT_POS,LOW);
  currentMode = 1;
}


void move_drum(int direction) {
  if (direction == 0){
    stopActuators();
    return;
  }
  else if (direction == 1) {
    if (mining) {
      actuator_pulse_limit = 1800;
      analogWrite(MINE_LEFT_SPEED, miningSpeed);
      analogWrite(MINE_RIGHT_SPEED, miningSpeed);
    }
    else if (depositing) {
      actuator_pulse_limit = 1200;
      analogWrite(MINE_LEFT_SPEED, depositSpeed);
      analogWrite(MINE_RIGHT_SPEED, depositSpeed);
    }
    isMovingForward = true;
    digitalWrite(MINE_RIGHT_NEG, LOW);
    digitalWrite(MINE_LEFT_NEG, LOW);
    digitalWrite(MINE_RIGHT_POS, HIGH);
    digitalWrite(MINE_LEFT_POS, HIGH);

    

    lastPulseTime = millis(); 

    noInterrupts();
    lastPulseCountLeft = pulseCountLeft;
    interrupts();  
    while(isActuatorMoving()) {
      delay(50);
    }
    depositing = true;
    mining = false;
  }
  else if (direction == 2) {
    isMovingForward = false;
    analogWrite(MINE_RIGHT_SPEED, 255);
    analogWrite(MINE_LEFT_SPEED, 255);
    digitalWrite(MINE_RIGHT_POS, LOW);
    digitalWrite(MINE_LEFT_POS, LOW);
    digitalWrite(MINE_RIGHT_NEG, HIGH);
    digitalWrite(MINE_LEFT_NEG, HIGH);

    lastPulseTime = millis(); 

    noInterrupts();
    lastPulseCountLeft = pulseCountLeft;
    interrupts();
    while(isActuatorMoving()) {
      delay(50);
    }
    pulseCountRight = 0; pulseCountLeft = 0;
  }
}

void stopActuators() {
  digitalWrite(D_ACT_POS, LOW);
  digitalWrite(D_ACT_NEG, LOW);
  digitalWrite(MINE_RIGHT_POS, LOW);
  digitalWrite(MINE_RIGHT_NEG, LOW);
  digitalWrite(MINE_LEFT_POS, LOW);
  digitalWrite(MINE_LEFT_NEG, LOW);
}

// --- Motor Logic ---

// Converts -1.0 to 1.0 into 0 to 180 for SparkMax
int mapSpeed(float val) {
  return map(val * 100, -100, 100, 45,135); // changed because of motor opp. installation
}

void run_drum(int action, float val) {
  if (action == 1) miningMotor.write(mapSpeed(val));
  else if (action == 2) miningMotor.write((180 - mapSpeed(val)));
  else miningMotor.write(90);
}

void accelerate(float targetSpeed) {
  float accelStep = 0.01;
  
  while (abs(currentSpeed - targetSpeed) > 0.01) {
    if (currentSpeed < targetSpeed) currentSpeed += accelStep;
    else currentSpeed -= accelStep;

    // Clamp speed
    if (currentSpeed > 1.0) currentSpeed = 1.0;
    if (currentSpeed < -1.0) currentSpeed = -1.0;

    if (currentMode == 1) executeLinear(currentSpeed);
    else if (currentMode == 2) executePointTurn(currentSpeed);
    
    delay(20);
  }
}

void executeLinear(float val) {
  //stopMotors();
  int servoVal = mapSpeed(val);
  leftMotor.write((180 - servoVal));
  rightMotor.write(servoVal);
  //adelay(5000);
}

void executePointTurn(float val) {
//motors are installed in opposite directions: if both motors at the same value they spin opposite
  //stopMotors();
  int leftVal = mapSpeed(val);
  int rightVal = mapSpeed(val);
  leftMotor.write((180 - (leftVal)));
  rightMotor.write((180 - (rightVal)));
  //delay(5000);
}

void stopMotors() {
  currentSpeed = 0;
  leftMotor.write(90);
  rightMotor.write(90);
  miningMotor.write(90);
}


//Interupt function
void countPulse() {
  // Counts up if extending, counts down if retracting
 if (isMovingForward && pulseCountRight < actuator_pulse_limit && pulseCountLeft < actuator_pulse_limit) {
    pulseCountRight++; pulseCountLeft++;

  } else if (!isMovingForward) {
    pulseCountRight--; pulseCountLeft--;
  }
  if (pulseCountRight >= actuator_pulse_limit || pulseCountLeft >= actuator_pulse_limit) {
    stopActuators();
  }
}

bool isActuatorMoving() {
  noInterrupts();
  long currentPulses = pulseCountLeft;
  interrupts();

  // If the pulse count changed, reset the timer
  if (currentPulses != lastPulseCountLeft) {
    lastPulseCountLeft = currentPulses;
    lastPulseTime = millis();
    return true; // It is still moving
  }

  // If it hasn't pulsed in 250ms, it has hit the limit switch and stalled
  if (millis() - lastPulseTime > 250) {
    return false; // It stopped moving
  }

  return true; // It hasn't pulsed yet, but we haven't hit the 250ms timeout either
}
